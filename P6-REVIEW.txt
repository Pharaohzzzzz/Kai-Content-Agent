"""Render a versioned original storyboard and local narration to a review candidate."""

import argparse
from pathlib import Path
import shutil
import subprocess
import sys
from tempfile import TemporaryDirectory
from time import perf_counter

from analysis_plan import json_text, read_json, require, required_text
from create_audio import RIGHTS, local_asset, prepare_audio
from create_visuals import asset_region, captions, scene_events
from media import run_media_tool
from render import ENCODERS, inspect_video, new_output, sha256
from render_brand import DEFAULT_BRAND, FPS, color_input, ffmpeg_still, load_brand
from render_commentary import SAMPLE_RATE
from render_subtitles import write_events

HUMAN_REVIEW = "pending"


def disclosure(plan):
    return "含 AI 生成内容" if plan["ai_generated"] else ""


def load_storyboard(path, brand):
    plan = read_json(path)
    require(type(plan["schema_version"]) is int and plan["schema_version"] == 1, "Expected storyboard schema version 1.")
    for label in ("topic_id", "title", "evidence"):
        required_text(plan[label], label)
    review_note = (path.parent / plan["evidence"]).resolve()
    require(review_note.is_file() and review_note.suffix.lower() == ".md", "Use a local Markdown evidence record.")
    require(plan["mode"] in {"release-candidate", "method-demo-draft"}, "Choose a supported production mode.")
    require(type(plan["ai_generated"]) is bool, "Declare the AI content status.")
    if plan["ai_generated"]:
        required_text(plan["ai_note"], "AI content note")
    voice_mode = plan["voice"]["mode"]
    require(voice_mode in {"windows-tts", "recording"}, "Provide Windows TTS or original recordings.")
    require(voice_mode != "windows-tts" or plan["ai_generated"], "Mark synthesized narration as AI content.")

    require(isinstance(plan["sources"], list), "Sources must be a list of source records.")
    for source in plan["sources"]:
        for key in ("title", "url", "use"):
            required_text(source[key], f"source {key}")
    require(isinstance(plan["materials"], list) and plan["materials"], "Record the origin and rights of the visual and audio materials.")
    for item in plan["materials"]:
        for key in ("description", "origin", "rights_note"):
            required_text(item[key], f"material {key}")
        require(item["rights"] in RIGHTS, f"Unconfirmed material rights: {item['description']}")
        require(type(item["ai_generated"]) is bool and (not item["ai_generated"] or plan["ai_generated"]),
                f"Check the AI declaration for {item['description']}.")

    scenes = plan["scenes"]
    require(isinstance(scenes, list) and scenes and scenes[0]["visual"]["kind"] == "title", "A storyboard starts with a title scene.")
    ids = [required_text(scene["id"], "scene id") for scene in scenes]
    require(len(set(ids)) == len(ids), "Scene IDs must be unique.")
    frame_ids = plan.get("review_scene_ids", ids[:3])
    require(isinstance(frame_ids, list) and len(frame_ids) == len(set(frame_ids))
            and 1 <= len(frame_ids) <= 3 and set(frame_ids) <= set(ids), "Choose up to three distinct review scene IDs.")
    assets = {}
    for scene in scenes:
        narration = scene["narration"]
        require(isinstance(narration, list) and narration, f"Scene {scene['id']} needs audible narration.")
        for cue in narration:
            spoken = required_text(cue["spoken"], "spoken phrase")
            caption = required_text(cue.get("subtitle", spoken), "subtitle")
            captions(brand, [dict(text=caption, start_ms=0, end_ms=1000)])
            if voice_mode == "recording":
                local_asset(path.parent, cue["recording"], "narration recording")
                require(not cue["recording"]["ai_generated"] or plan["ai_generated"], "Recording AI declaration is missing.")
        visual = scene["visual"]
        required_text(visual["title"], "visual title")
        if visual["kind"] in {"image", "video"}:
            assets[scene["id"]] = local_asset(path.parent, visual["asset"], "scene asset")
            require(not visual["asset"]["ai_generated"] or plan["ai_generated"], "Visual asset AI declaration is missing.")
        scene_events(brand, visual, 1000, disclosure(plan), draft=plan["mode"] == "method-demo-draft")
    cover = {**scenes[0]["visual"], "title": plan["title"]}
    scene_events(brand, cover, 1000, disclosure(plan), cover=True, draft=plan["mode"] == "method-demo-draft")
    return plan, review_note, assets, frame_ids


def draw_scene(directory, brand, visual, asset, frame_count, media_name, cover=False, encoder="h264_nvenc"):
    inputs = ["-f", "lavfi", "-i", color_input(brand)]
    graph = "[0:v]format=yuv420p"
    if asset:
        region = asset_region(brand)
        w, h = region["w"], region["h"]
        inputs += ["-i", str(asset)]
        graph += (
            f"[base];[1:v:0]setpts=PTS-STARTPTS,fps={FPS},"
            f"scale={w}:{h}:force_original_aspect_ratio=decrease:force_divisible_by=2:reset_sar=1"
            f"[asset];[base][asset]overlay="
            f"x={region['x']}+({w}-w)/2:y={region['y']}+({h}-h)/2:eof_action=repeat:repeatlast=1"
        )
    graph += ",ass=scene.ass[v]"
    args = [*inputs, "-filter_complex_threads", "1", "-filter_complex", graph, "-map", "[v]"]
    if cover:
        ffmpeg_still([*args, "-frames:v", "1", "-threads:v", "1", "-update", "1", media_name], directory)
        return
    result = subprocess.run([
        "ffmpeg", "-hide_banner", "-loglevel", "info", "-nostdin", "-n", "-nostats", "-xerror",
        *args, "-frames:v", str(frame_count), "-an", "-c:v", encoder, *ENCODERS[encoder],
        "-bf", "0", "-pix_fmt", "yuv420p", "-video_track_timescale", str(FPS * 512), media_name,
    ], cwd=directory, capture_output=True, text=True, encoding="utf-8", errors="replace")
    require(result.returncode == 0, f"Scene render failed: {visual['title']}\n" + "\n".join(result.stderr.splitlines()[-18:]))
    require("failed to find any fallback" not in result.stderr.lower(), "A scene glyph is missing in libass.")
    require("fontselect:" in result.stderr, "libass did not select a scene font.")


def render(args):
    started = perf_counter()
    source = args.storyboard.resolve()
    brand, brand_sha = load_brand(args.brand.resolve())
    plan, review_note, assets, frame_ids = load_storyboard(source, brand)
    if args.check:
        print(f"VALID original storyboard: {plan['title']} | {', '.join(scene['id'] for scene in plan['scenes'])}")
        return
    require(args.output is not None, "Rendering needs a new --output under the project output directory.")
    output = new_output(args.output)
    output.parent.mkdir(parents=True, exist_ok=True)
    with TemporaryDirectory(prefix=".create-", dir=output.parent) as temp:
        scratch = Path(temp)
        staged = scratch / "ready"
        staged.mkdir()
        voice_start = perf_counter()
        timeline, voice = prepare_audio(plan, source.parent, scratch)
        voice_seconds = perf_counter() - voice_start
        encode_start = perf_counter()
        scene_files, all_captions = [], []
        label = disclosure(plan)
        draft = plan["mode"] == "method-demo-draft"
        for index, (scene, span) in enumerate(zip(plan["scenes"], timeline, strict=True)):
            duration = span["end_ms"] - span["start_ms"]
            frames = span["end_frame"] - span["start_frame"]
            events = scene_events(brand, scene["visual"], duration, label, draft=draft)
            events += captions(brand, span["cues"], span["start_ms"])
            all_captions.extend(captions(brand, span["cues"]))
            write_events(scratch / "scene.ass", brand, events)
            name = f"scene-{index:03}.mp4"
            draw_scene(scratch, brand, scene["visual"], assets.get(scene["id"]), frames, name, encoder=args.encoder)
            scene_files.append(f"file '{name}'\nduration {frames / FPS:.9f}\n")

        cuts = scratch / "scenes.ffconcat"
        cuts.write_text("ffconcat version 1.0\n" + "".join(scene_files), encoding="ascii")
        run_media_tool([
            "ffmpeg", "-v", "error", "-xerror", "-nostdin", "-n", "-f", "concat", "-safe", "1",
            "-i", str(cuts), "-i", str(scratch / "narration.wav"), "-map", "0:v:0", "-map", "1:a:0",
            "-c:v", "copy", "-c:a", "aac", "-b:a", "128k", "-ar", str(SAMPLE_RATE), "-ac", "1",
            "-map_metadata", "-1", "-movflags", "+faststart", str(staged / "video.mp4"),
        ])
        encode_seconds = perf_counter() - encode_start
        media_info = inspect_video(staged / "video.mp4", timeline[-1]["end_frame"], 1)
        write_events(staged / "subtitles.ass", brand, all_captions)

        intro = {**plan["scenes"][0]["visual"], "title": plan["title"]}
        write_events(scratch / "scene.ass", brand, scene_events(brand, intro, 1000, label, cover=True, draft=draft))
        draw_scene(scratch, brand, intro, None, 1, "cover.png", cover=True)
        shutil.move(scratch / "cover.png", staged / "cover.png")
        spans = {scene["id"]: span for scene, span in zip(plan["scenes"], timeline, strict=True)}
        frame_records = []
        for index, scene_id in enumerate(frame_ids, 1):
            cue = max(spans[scene_id]["cues"], key=lambda item: item["end_ms"] - item["start_ms"])
            at_ms = (cue["start_ms"] + cue["end_ms"]) // 2
            filename = f"frame-{index:02}.png"
            ffmpeg_still([
                "-ss", f"{at_ms / 1000:.3f}", "-i", str(staged / "video.mp4"), "-map", "0:v:0",
                "-frames:v", "1", "-threads:v", "1", "-update", "1", filename,
            ], staged)
            frame_records.append(dict(file=filename, scene_id=scene_id, at_ms=at_ms))

        evidence = "facts-and-test.md"
        shutil.copyfile(review_note, staged / evidence)
        snapshot = {
            **plan, "evidence": evidence, "source_storyboard": str(source), "human_review": HUMAN_REVIEW,
            "scenes": [{**scene, "timeline": span} for scene, span in zip(plan["scenes"], timeline, strict=True)],
        }
        (staged / "storyboard.json").write_text(json_text(snapshot), encoding="utf-8")
        manifest = dict(
            schema_version=1, title=plan["title"], topic_id=plan["topic_id"], mode=plan["mode"],
            human_review=HUMAN_REVIEW, ai_generated=plan["ai_generated"], ai_note=plan.get("ai_note"),
            storyboard=dict(path=str(source), sha256=sha256(source)),
            brand=dict(path=str(args.brand.resolve()), sha256=brand_sha, version=brand["version"]),
            source_assets={key: dict(path=str(value), sha256=sha256(value)) for key, value in assets.items()},
            voice=voice, encoder=args.encoder, media=media_info,
            timings=dict(narration_seconds=round(voice_seconds, 3), encoding_seconds=round(encode_seconds, 3),
                         production_seconds=round(perf_counter() - started, 3)),
            review_frames=frame_records,
            artifacts_sha256={file.name: sha256(file) for file in sorted(staged.iterdir())},
        )
        (staged / "render_manifest.json").write_text(json_text(manifest), encoding="utf-8")
        staged.rename(output)
    print(f"KAI ORIGINAL | {output} | {media_info['duration_seconds']:.3f}s | human review pending")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("storyboard", type=Path)
    parser.add_argument("--output", type=Path)
    parser.add_argument("--brand", type=Path, default=DEFAULT_BRAND)
    parser.add_argument("--encoder", choices=ENCODERS, default="h264_nvenc")

    except (OSError, RuntimeError, TypeError, ValueError, KeyError) as error:

    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Prepare exact phrase and scene timing from Windows TTS or original recordings."""

from array import array
from pathlib import Path
import subprocess
import sys
import wave

from analysis_plan import json_text, require, required_text
from media import run_media_tool
from render import sha256
from render_brand import FPS
from render_commentary import SAMPLE_RATE

SAMPLES_PER_FRAME = SAMPLE_RATE // FPS
WAV_FORMAT = (1, 2, SAMPLE_RATE)
DEFAULT_PAUSE_MS = 110
RIGHTS = {"owned", "licensed", "permission-granted"}


def local_asset(base, item, label):
    path = (base / required_text(item["file"], label)).resolve()
    require(path.is_file(), f"{label} is missing: {path}")
    require(item["rights"] in RIGHTS, f"Record confirmed rights for {label}: {path}")
    required_text(item["rights_note"], f"{label} rights note")
    require(type(item["ai_generated"]) is bool, f"Declare AI status for {label}.")
    return path


def speech_wav(source, target, clip):
    trim = []
    if clip is not None:
        start, end = clip
        require(type(start) is int and type(end) is int and 0 <= start < end, "Invalid recording clip_ms.")
        trim = ["-af", f"atrim=start={start / 1000:.3f}:end={end / 1000:.3f},asetpts=PTS-STARTPTS"]
    run_media_tool([
        "ffmpeg", "-v", "error", "-xerror", "-nostdin", "-n", "-i", str(source),
        "-map", "0:a:0", *trim, "-ar", str(SAMPLE_RATE), "-ac", "1", "-c:a", "pcm_s16le", str(target),
    ])
    if clip is not None:
        with wave.open(str(target), "rb") as audio:
            require(abs(audio.getnframes() - (end - start) * SAMPLE_RATE // 1000) <= 960,
                    f"Recording ends before the specified clip: {source}")


def voiced_pcm(path):
    with wave.open(str(path), "rb") as audio:
        actual_format = (audio.getnchannels(), audio.getsampwidth(), audio.getframerate())
        require(actual_format == WAV_FORMAT, f"Unexpected narration WAV format: {path}")
        pcm = audio.readframes(audio.getnframes())
    samples = array("h")
    samples.frombytes(pcm)
    if sys.byteorder != "little":
        samples.byteswap()
    window = SAMPLE_RATE // 100
    active = []
    for index in range(0, len(samples), window):
        block = samples[index:index + window]
        if sum(sample * sample for sample in block) >= 110 ** 2 * len(block):
            active.append(index)
    require(len(active) >= 8, f"No usable speech in {path}. Supply an audible recording or another local voice.")
    # Keep quiet consonants just outside the measured energy span. Internal pauses stay intact.
    start = max(0, active[0] - 4 * window)
    end = min(len(samples), active[-1] + 5 * window)
    return pcm[2 * start:2 * end]


def prepare_audio(plan, base, directory):
    config = plan["voice"]
    mode = config["mode"]
    parts = [cue for scene in plan["scenes"] for cue in scene["narration"]]
    voice = {"mode": mode}
    if mode == "windows-tts":
        voice_name = required_text(config["voice"], "Windows voice")
        rate = config.get("rate", 0)
        require(type(rate) is int and -10 <= rate <= 10, "Windows speech rate must be an integer from -10 to 10.")
        request = directory / "tts.json"
        request.write_text(json_text({
            "voice": voice_name, "rate": rate, "parts": [part["spoken"] for part in parts],
        }), encoding="utf-8")
        script = Path(__file__).with_name("create_voice.ps1").resolve()
        result = subprocess.run([
            "powershell.exe", "-NoProfile", "-NonInteractive", "-File", str(script), str(request), str(directory),
        ], capture_output=True, text=True, encoding="utf-8", errors="replace")
        require(result.returncode == 0, f"Chinese TTS failed. Use a working Windows voice or import original recordings.\n{result.stderr}")
        voice.update(engine="Windows System.Speech", voice=voice_name, rate=rate)

    def ms(sample):
        return round(sample * 1000 / SAMPLE_RATE)

    cursor = 0
    part_index = 0
    input_hashes = {}
    timeline = []
    with wave.open(str(directory / "narration.wav"), "wb") as joined:
        joined.setnchannels(1)
        joined.setsampwidth(2)
        joined.setframerate(SAMPLE_RATE)
        for scene in plan["scenes"]:
            scene_start = cursor
            cues = []
            for cue in scene["narration"]:
                if mode == "windows-tts":
                    wav = directory / f"tts-{part_index:04}.wav"
                else:
                    source = local_asset(base, cue["recording"], "recording")
                    if source not in input_hashes:
                        input_hashes[source] = sha256(source)
                    wav = directory / "converted-voice.wav"
                    speech_wav(source, wav, cue["recording"].get("clip_ms"))
                pcm = voiced_pcm(wav)
                count = len(pcm) // 2
                require(count >= SAMPLE_RATE // 5, f"Speech cue is too short: {cue['spoken']}")
                joined.writeframesraw(pcm)
                end = cursor + count
                cues.append(dict(text=cue.get("subtitle", cue["spoken"]), start_ms=ms(cursor), end_ms=ms(end),
                                 start_sample=cursor, end_sample=end))
                pause = cue.get("pause_ms", DEFAULT_PAUSE_MS)
                require(type(pause) is int and 0 <= pause <= 600, "Cue pause_ms must be an integer between 0 and 600.")
                silence = pause * SAMPLE_RATE // 1000
                joined.writeframesraw(b"\0\0" * silence)
                cursor = end + silence
                if mode == "recording":
                    wav.unlink()
                part_index += 1
            frame_padding = -cursor % SAMPLES_PER_FRAME
            joined.writeframesraw(b"\0\0" * frame_padding)
            cursor += frame_padding
            timeline.append(dict(
                start_frame=scene_start // SAMPLES_PER_FRAME, end_frame=cursor // SAMPLES_PER_FRAME,
                start_ms=ms(scene_start), end_ms=ms(cursor), cues=cues,
            ))
    voice["recording_sha256"] = {str(path): checksum for path, checksum in input_hashes.items()}
    voice["timing"] = "Energy crop with 40 ms guard on each phrase, then measured PCM samples; no word-level timing."
    return timeline, voice

"""Compose data-driven original scenes on the Kai portrait canvas."""

from analysis_plan import require
from render_brand import HEIGHT, WIDTH
from render_subtitles import ass_color, block_height, events_for_block, fit_text, timestamp


def area(x, y, w, h):
    return dict(x=x, y=y, w=w, h=h)


def captions(brand, cues, offset_ms=0):
    safe = brand["platform_ui_reserve"]
    x = safe["left"] + 16
    right = WIDTH - safe["right"] - 16
    bottom = HEIGHT - safe["bottom"] - 32
    box = area(x, bottom - 130, right - x, 130)
    events = []
    for cue in cues:
        style = {**brand["styles"]["Chinese"], "width_factor": 1.0}
        block = fit_text(cue["text"], box, style, preferred_lines=2)
        require(len(block["lines"]) <= 2 and block_height(block) <= box["h"],
                f"Split this subtitle into shorter spoken cues: {cue['text']}")
        for line in events_for_block("Chinese", box, cue["start_ms"] - offset_ms,
                                     cue["end_ms"] - offset_ms, block):
            events.append(line.replace("Dialogue: 0,", "Dialogue: 2,", 1)
                          .replace(r"{\an7", r"{\an7\bord4\shad0", 1))
    return events


def asset_region(brand):
    safe = brand["platform_ui_reserve"]
    x = safe["left"] + 16
    return area(x, 490, WIDTH - safe["right"] - 10 - x, HEIGHT - safe["bottom"] - 760)


def scene_events(brand, visual, duration_ms, disclosure, *, cover=False, draft=False):
    safe, colors = brand["platform_ui_reserve"], brand["colors"]
    left = safe["left"] + 16
    right = WIDTH - safe["right"] - 10
    width = right - left
    end = timestamp(duration_ms)
    events = []

    def rect(x, y, w, h, color, alpha="00"):
        if w <= 0 or h <= 0:
            return
        fill = ass_color(color)[4:]
        tags = rf"{{\an7\pos({x},{y})\p1\bord0\shad0\1c&H{fill}&\1a&H{alpha}&}}"
        shape = f"m 0 0 l {w} 0 {w} {h} 0 {h}"
        events.append(f"Dialogue: 0,0:00:00.00,{end},Brand,,0,0,0,,{tags}{shape}\n")

    def text(value, x, y, w, style_name, size, lines=1, color=None):
        if not value:
            return
        style = {**brand["styles"][style_name], "size": size, "line_height": round(size * 1.22), "width_factor": 1.0}
        box = area(x, y, w, round(size * 1.22) * lines)
        block = fit_text(value, box, style, preferred_lines=lines)
        require(len(block["lines"]) <= lines, f"Shorten this scene text or split it across lines: {value}")
        for event in events_for_block(style_name, box, 0, duration_ms, block):
            if color:
                event = event.replace(r"{\an7", rf"{{\an7\1c&H{ass_color(colors[color])[4:]}&", 1)
            events.append(event.replace("Dialogue: 0,", "Dialogue: 1,", 1))

    def line(y, alpha="A0"):
        rect(left, y, width, 2, colors["accent"], alpha)

    kind = visual["kind"]
    top = max(safe["top"] + 16, 298 if cover else 156)
    text("KAI", left, top, width // 2, "Brand", 34)
    if draft:
        text("方法演示草稿", right - 273, top + 3, 273, "Brand", 26)
    if disclosure:
        text(disclosure, right - 273, top + (37 if draft else 5), 273, "Brand", 26, color="secondary")
    line(top + (80 if draft else 63), "45")
    text(visual.get("label"), left, top + 116, width, "CardLabel", 32)

    if kind == "title":
        text(visual["title"], left, top + 188, width, "CoverTitle", 100, 3)
        band_y = top + (555 if cover else 665)
        rect(left, band_y, width, 298, colors["accent"])
        text(visual["headline"], left + 34, band_y + 38, width - 68, "CardText", 66, 2, "background")
        text(visual.get("detail"), left + 36, band_y + 222, width - 72, "Brand", 32, 1, "background")
        text(visual.get("footer"), left, band_y + 353, width, "Brand", 31, 2, "secondary")
        return events

    text(visual["title"], left, top + 183, width, "Title", 64, 2)
    content_y = top + 350
    if kind == "chart":
        bars = visual["bars"]
        require(1 <= len(bars) <= 3, "A chart needs one to three bars.")
        spacing = 255 if len(bars) < 3 else 215
        for index, bar in enumerate(bars):
            value = bar["value"]
            require(type(value) in (int, float) and 0 <= value <= 100, "Chart values must be percentages from 0 to 100.")
            y = content_y + index * spacing
            text(bar["label"], left, y + 29, width - 247, "Chinese", 39, 2)
            text(f"{value:g}%", right - 236, y, 236, "Title", 63, color="accent")
            rect(left, y + 119, width, 38, colors["secondary"], "C0")
            rect(left, y + 119, round(width * value / 100), 38, colors["accent"], "42" if index == 0 else "00")
        foot_y = content_y + len(bars) * spacing + 15

    elif kind == "table":
        columns, rows = visual["columns"], visual["rows"]
        require(2 <= len(columns) <= 4 and 1 <= len(rows) <= 6
                and all(len(row) == len(columns) for row in rows), "Use 2–4 columns and 1–6 matching rows.")
        weights = visual.get("column_weights", [1] * len(columns))
        require(len(weights) == len(columns) and all(type(item) is int and item > 0 for item in weights),
                "Column weights must be positive integers, one per column.")
        widths = [width * weight // sum(weights) for weight in weights]
        row_height = min(119, 590 // (len(rows) + 1))
        highlight = set(visual.get("highlight_rows", []))
        require(all(type(index) is int and 1 <= index <= len(rows) for index in highlight),
                "Highlight rows use 1-based indexes.")
        for index, row in enumerate([columns, *rows]):
            y = content_y + index * row_height
            if index == 0 or index in highlight:
                rect(left, y, width, row_height, colors["accent"], "C8" if index == 0 else "DC")
            line(y + row_height - 1, "C0")
            x = left
            for cell, cell_width in zip(row, widths, strict=True):
                text(cell, x + 15, y + 19, cell_width - 27, "Chinese", 34 if index == 0 else 39,
                     1, "accent" if index == 0 or index in highlight else None)
                x += cell_width
        foot_y = content_y + (len(rows) + 1) * row_height + 45

    elif kind == "conclusion":
        line(content_y)
        items = visual["items"]
        require(1 <= len(items) <= 3, "A conclusion needs one to three takeaways.")
        for index, item in enumerate(items):
            y = content_y + 49 + index * 200
            rect(left, y + 5, 7, 112, colors["accent"])
            text(item, left + 33, y, width - 44, "CardText", 57, 2)
        foot_y = content_y + 49 + len(items) * 200 - 15

    elif kind in {"image", "video"}:
        region = asset_region(brand)
        foot_y = region["y"] + region["h"] + 20

    else:
        raise ValueError(f"Unknown visual kind: {kind}")

    text(visual.get("footer"), left, foot_y, width, "Brand", 30, 2, "secondary")
    return events

===== src/create_voice.ps1 =====

param(
    [Parameter(Mandatory = $true)][string]$Request,
    [Parameter(Mandatory = $true)][string]$Directory
)

$ErrorActionPreference = 'Stop'
Add-Type -AssemblyName System.Speech
$job = Get-Content -LiteralPath $Request -Raw -Encoding UTF8 | ConvertFrom-Json
$synth = [System.Speech.Synthesis.SpeechSynthesizer]::new()
try {
    $synth.SelectVoice($job.voice)
    if ($synth.Voice.Culture.Name -notlike 'zh-*') {
        throw "The selected Windows voice is not Chinese."
    }
    $synth.Rate = $job.rate
    $format = [System.Speech.AudioFormat.SpeechAudioFormatInfo]::new(
        48000, [System.Speech.AudioFormat.AudioBitsPerSample]::Sixteen,
        [System.Speech.AudioFormat.AudioChannel]::Mono
    )
    for ($index = 0; $index -lt $job.parts.Count; $index++) {
        $name = 'tts-{0:D4}.wav' -f $index
        $synth.SetOutputToWaveFile((Join-Path $Directory $name), $format)
        $synth.Speak([string]$job.parts[$index])
        $synth.SetOutputToNull()
    }
    $synth.Voice.Name
} finally {
    $synth.Dispose()
}

# 原创场景数据格式 V1

`src/create.py` 读取 UTF-8 JSON。场景和句子按数组顺序播放。脚本、数字、标题、颜色配置和素材都可独立更换。

| 顶层字段 | 用法 |
| --- | --- |
| `schema_version` | `1` |
| `topic_id`, `title` | 选题唯一 ID 与封面标题 |
| `mode` | `release-candidate` 或 `method-demo-draft` |
| `ai_generated`, `ai_note` | 布尔标识。值为真时写明哪些内容涉及 AI |
| `voice` | `{"mode":"windows-tts","voice":"Microsoft Huihui Desktop","rate":1}`，或 `{"mode":"recording"}` |
| `evidence` | 故事板目录下的 Markdown 路径，写事实和实测摘要 |
| `sources` | 数组，条目含 `title`、`url`、`use`。无外部事实时可为空 |
| `materials` | 非空数组，条目含 `description`、`origin`、`rights`、`rights_note`、`ai_generated` |
| `scenes` | 条目含唯一 `id`、一个 `visual` 对象和非空 `narration` 数组 |
| `review_scene_ids` | 可选，1 到 3 个场景 ID，按指定顺序出视频帧 |

每个 `visual` 有 `kind`、`title`，可加 `label`、`footer`。同种场景可出现多次。长标题或表格应先运行 `--check`。

| 类型 | 字段 |
| --- | --- |
| `title` | 首个场景必用。必填 `headline`，可加 `detail` |
| `chart` | `bars` 含 1 到 3 个 `{"label":"条目","value":85.8}`，单位为百分比 |
| `table` | `columns` 为 2 到 4 个文字列名，`rows` 为 1 到 6 行文字数组。可给 `column_weights` 正整数数组和 `highlight_rows` 行号数组（从 1 起） |
| `conclusion` | `items` 含 1 到 3 条简短结论 |
| `image` / `video` | `asset` 声明已授权的本地图片或录屏。录屏从头播放，短素材定格末帧，声音使用独立口播 |

`narration` 每条填 `spoken`。可设 `subtitle` 以数字等书面写法匹配实际口播，默认使用 `spoken`。短句之间可设整数 `pause_ms`，默认 110 毫秒。录音模式每条另填 `recording`。可提交一句一个文件，或在同一文件上用 `clip_ms:[开始,结束]` 声明人工校对的范围。素材和录音记录均包含 `file`、`rights`、`rights_note`、`ai_generated`。相对路径从输入故事板所在目录解析。授权值为 `owned`、`licensed` 或 `permission-granted`。

场景片段随口播样本长度生成，补齐到视频帧边界。产物 `storyboard.json` 记录每场及每句样本或毫秒边界，`subtitles.ass` 提供中文字幕。精确到逐词的时间需另有实测依据。产物故事板是审核快照，重制用原始故事板与其相对路径素材。
