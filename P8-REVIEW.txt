"""P8 command line for recording actual publications and reviewing results."""

import argparse
import csv
import json
import sys
from pathlib import Path

from review_data import FORMATS, PLATFORMS, Operations, confirm, import_metrics, register
from review_report import report


def production_fields(command):
    command.add_argument("--manual-minutes", help="Human production minutes allocated to this publication")
    command.add_argument("--production-cost-cny", help="Production cost allocated to this publication (CNY)")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--simulate", action="store_true", help="Use the isolated simulated workspace")
    commands = parser.add_subparsers(dest="command", required=True)

    pending = commands.add_parser("register", help="Associate a local P7 package with pending records")
    pending.add_argument("manifest", type=Path)
    pending.add_argument("--category", required=True)
    pending.add_argument("--platform", choices=PLATFORMS)
    pending.add_argument("--format", choices=FORMATS)
    production_fields(pending)

    actual = commands.add_parser("confirm", help="Record actual publication after the user confirms it")
    actual.add_argument("publication_id")
    actual.add_argument("--confirmed-published", action="store_true")
    actual.add_argument("--title", required=True)
    actual.add_argument("--published-at", required=True)
    actual.add_argument("--work-id")
    actual.add_argument("--url")
    production_fields(actual)

    commands.add_parser("list", help="List the pending and published records")
    metrics = commands.add_parser("import", help="Import cumulative metrics from CSV or JSONL")
    metrics.add_argument("file", type=Path)

    brief = commands.add_parser("report", help="Create a Markdown and JSON operation review")
    brief.add_argument("--start", help="First publication date, inclusive (Asia/Shanghai)")
    brief.add_argument("--through", help="Last publication date, inclusive (Asia/Shanghai)")
    brief.add_argument("--as-of", help="Use snapshots through this ISO 8601 time (default: now)")
    brief.add_argument("--name", help="Report folder name")
    brief.add_argument("--editorial", type=Path, help="Codex notes for this report's current numerical basis")

    args = parser.parse_args()
    operations = Operations(args.simulate)
    try:
        with operations.lock:
            if args.command == "report":
                print(report(args, operations))
                return 0
            if args.command == "list":
                _, _, records = operations.publications()
                result = [record for record, _ in records.values()]
            else:
                handler = {"register": register, "confirm": confirm, "import": import_metrics}[args.command]
                result = handler(args, operations)
            print(json.dumps(result, ensure_ascii=False, indent=2, allow_nan=False))
    except (OSError, ValueError, KeyError, AttributeError, TypeError, csv.Error) as error:
        print(f"Review error: {error}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

def timestamp(value, label):
    try:
        stamp = datetime.fromisoformat(required_text(value, label))
        if stamp.utcoffset() is None:
            raise ValueError()
    except ValueError as error:
        raise ValueError(f"{label} requires ISO 8601 with a timezone offset") from error
    return stamp.astimezone(timezone.utc)


def optional_text(value, label):
    return None if value is None or isinstance(value, str) and not value.strip() else required_text(value, label)


def number(value, label, *, integer=False, money=False, rate=False):
    if value is None or isinstance(value, str) and not value.strip():
        return None
    try:
        raw = str(value).strip()
        percent = rate and raw.endswith("%")
        parsed = Decimal(raw[:-1] if percent else raw)
        ceiling = 10**12 if money else 10**15
        if not parsed.is_finite() or not 0 <= parsed <= ceiling:
            raise InvalidOperation()
        if percent:
            parsed /= 100
        if (rate and parsed > 1 or integer and parsed != parsed.to_integral_value()
                or money and parsed != parsed.quantize(Decimal("0.01"))):
            raise InvalidOperation()
    except InvalidOperation as error:
        unit = "a fraction 0–1 or a percentage" if rate else "a non-negative number"
        raise ValueError(f"{label} must be {unit} with the expected precision") from error
    return int(parsed) if integer else float(parsed)


def read_jsonl(path):
    rows = []
    with path.open(encoding="utf-8-sig") as source:
        for index, line in enumerate(source, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
                if not isinstance(row, dict):
                    raise ValueError("expected an object")
            except ValueError as error:
                raise ValueError(f"{path.name}, line {index}: {error}") from error
            rows.append((index, row))
    return rows


def write_jsonl(path, records):
    atomic_write(path, "".join(json.dumps(row, ensure_ascii=False, allow_nan=False) + "\n"
                               for row in records))

        self.snapshots = self.directory / "snapshots.jsonl"
        self.history = self.directory / "research-history.jsonl" if simulate else HISTORY
        self.lock = history_lock(self.history)

    def load(self, path):
        rows = [row for _, row in read_jsonl(path)] if path.exists() else []
        if any(row.get("data_mode") != self.mode for row in rows):
            raise ValueError(f"{path.name} contains a record from the wrong data mode")
        return rows

    def publications(self):
        history = read_history(self.history)
        standalone = self.load(self.standalone)
        by_id = {row["publication_id"]: (row, None) for row in standalone}
        for topic in history:
            for row in topic.get("publications", []):
                if row.get("data_mode") != self.mode:
                    raise ValueError("Research history contains a publication from the wrong data mode")
                by_id[row["publication_id"]] = (row, topic)
        return history, standalone, by_id

    def save(self, history, standalone, topic):
        if topic is not None:
            write_history(history, self.history)
        else:
            self.directory.mkdir(parents=True, exist_ok=True)
            write_jsonl(self.standalone, standalone)


def candidates(args, operations):
    path = args.manifest.resolve()
    base = ROOT / "output" if operations.mode == "real" else operations.directory
    if not path.is_relative_to(base) or (operations.mode == "real" and path.name != "publish_manifest.json"):
        raise ValueError(f"Use a {operations.mode} publication manifest under {base}")
    raw = path.read_bytes()
    manifest = json.loads(raw.decode("utf-8-sig"))
    if manifest.get("data_mode", "real") != operations.mode:
        raise ValueError("Publication manifest has the wrong data mode")
    if manifest.get("schema_version") != 1 or manifest.get("status") != "awaiting_human_review":
        raise ValueError("Use a complete P7 package marked awaiting_human_review")
    manifest_sha = hashlib.sha256(raw).hexdigest()
    source = manifest["source"]
    lane = source["lane"]
    topic_id = optional_text(source.get("topic_id"), "topic_id")
    clip_id = optional_text(source.get("clip_id"), "clip_id")
    if not (lane == "A" and clip_id or lane == "B" and topic_id):
        raise ValueError("Manifest must identify the A-line clip or B-line topic")
    cost = number(args.production_cost_cny, "production_cost_cny", money=True)
    minutes = number(args.manual_minutes, "manual_minutes")
    records = []
    for platform in PLATFORMS:
        if args.platform and platform != args.platform:
            continue
        package = manifest["platforms"].get(platform, {})
        cards = package.get("cards", {})
        artifacts = {"video": [package["files"]["video"]]} if package.get("files", {}).get("video") else {}
        if cards.get("status") == "packaged" and cards.get("files"):
            artifacts["note"] = cards["files"]
        for form, files in artifacts.items():
            if args.format and form != args.format:
                continue
            duration = number(source["media"]["duration_seconds"], "duration_seconds") if form == "video" else None
            if form == "video" and not duration:
                raise ValueError("Video duration must be positive")
            prefix = "pub" if operations.mode == "real" else "sim"
            records.append({
                "schema_version": 1, "data_mode": operations.mode,
                "publication_id": f"{prefix}-{platform}-{form}-{manifest_sha[:16]}",
                "status": "pending", "topic_id": topic_id, "clip_id": clip_id,
                "lane": lane, "category": required_text(args.category, "category"),
                "render_manifest": source["render"], "publish_manifest": {
                    "path": path.relative_to(ROOT).as_posix(), "sha256": manifest_sha,
                }, "artifacts": [{key: file[key] for key in ("path", "sha256")} for file in files],
                "platform": platform, "format": form, "duration_seconds": duration,
                "manual_minutes": minutes, "production_cost_cny": cost,
                "title": None, "published_at": None, "work_id": None, "url": None, "confirmed_at": None,
            })
    if not records:
        raise ValueError("The package does not contain the requested platform or format")
    return records


def fill_production(record, values):
    for key in ("manual_minutes", "production_cost_cny"):
        value = values.get(key)
        if value is None:
            continue
        if record[key] is not None and record[key] != value:
            raise ValueError(f"{key} is already recorded for {record['publication_id']}")
        record[key] = value


def register(args, operations):
    records = candidates(args, operations)
    history, standalone, by_id = operations.publications()
    topic_id = records[0]["topic_id"]
    lane = records[0]["lane"]
    topic = next((row for row in history if topic_id and row["topic_id"] == topic_id), None)
    if operations.mode == "real" and lane == "B" and topic is None:
        raise ValueError(f"Unknown B-line topic in P5 history: {topic_id}")
    if topic is not None and topic["line"] != lane:
        raise ValueError("Publication lane differs from the P5 topic lane")
    target = topic.setdefault("publications", []) if topic else standalone
    result = []
    changed = False
    for row in records:
        identity = row["publication_id"]
        if identity in by_id:
            prior, owner = by_id[identity]
            if (owner is not topic or prior["category"] != row["category"]
                    or prior["publish_manifest"]["sha256"] != row["publish_manifest"]["sha256"]):
                raise ValueError(f"Publication already registered with different metadata: {identity}")
            previous = prior.copy()
            fill_production(prior, row)
            changed |= prior != previous
            result.append(prior)
        else:
            target.append(row)
            result.append(row)
            changed = True
    if changed:
        operations.save(history, standalone, topic)
    return result


def confirm(args, operations):
    if not args.confirmed_published:
        raise ValueError("Explicit user confirmation of actual publication is required: --confirmed-published")
    history, standalone, by_id = operations.publications()
    if args.publication_id not in by_id:
        raise ValueError(f"Unknown {operations.mode} publication: {args.publication_id}")
    row, topic = by_id[args.publication_id]
    stamp = timestamp(args.published_at, "published_at")
    now = datetime.now(timezone.utc)
    if operations.mode == "real" and stamp > now:
        raise ValueError("Actual publication time cannot be in the future")
    work_id = optional_text(args.work_id, "work_id")
    url = checked_url(args.url) if args.url is not None else None
    if work_id is None and url is None:
        raise ValueError("Provide a platform work ID or URL")
    for prior, _ in by_id.values():
        if prior is row or prior["platform"] != row["platform"] or prior["status"] != "published":
            continue
        if (work_id and prior["work_id"] == work_id
                or url and prior["url"] and url_key(url) == url_key(prior["url"])):
            raise ValueError(f"Platform work already recorded as {prior['publication_id']}")
    details = {
        "title": required_text(args.title, "actual title"),
        "published_at": stamp.isoformat(), "work_id": work_id, "url": url,
    }
    previous = row.copy()
    if row["status"] == "published":
        for key, value in details.items():
            if value is not None and row[key] not in (None, value):
                raise ValueError(f"Confirmed publication already has a different {key}")
    if topic is not None:
        state = topic["status"]
        if state != "published" and "published" not in TOPIC_STATES[state]:
            raise ValueError(f"Record production in P5 history before confirming publication: {state}")
        if state != "published":
            topic.update(status="published", status_note=f"P8 已确认实际发布 {row['publication_id']}",
                         status_at=now.astimezone(SHANGHAI).isoformat(timespec="seconds"))
    row.update({key: value for key, value in details.items() if value is not None})
    row.update(status="published", confirmed_at=row["confirmed_at"] or now.isoformat(timespec="seconds"))
    fill_production(row, {
        "manual_minutes": number(args.manual_minutes, "manual_minutes"),
        "production_cost_cny": number(args.production_cost_cny, "production_cost_cny", money=True),
    })
    if row != previous:
        operations.save(history, standalone, topic)
    return row


def input_rows(path):
    if path.suffix.lower() == ".jsonl":
        yield from read_jsonl(path)
    elif path.suffix.lower() == ".csv":
        with path.open(encoding="utf-8-sig", newline="") as source:
            reader = csv.DictReader(source, strict=True)
            names = reader.fieldnames or []
            if len(names) != len(set(names)) or not {"publication_id", "collected_at"} <= set(names):
                raise ValueError("CSV requires unique headers including publication_id and collected_at")
            if set(names) - set(CSV_FIELDS):
                raise ValueError(f"Unknown CSV columns: {', '.join(sorted(set(names) - set(CSV_FIELDS)))}")
            for row in reader:
                yield reader.line_num, row
    else:
        raise ValueError("Use a .csv or .jsonl file")


def snapshot(raw, operations, by_id, now):
    if set(raw) - {*METADATA, *METRICS, "data_mode"}:
        raise ValueError("Unknown metrics fields or extra CSV columns")
    if raw.get("data_mode", operations.mode) != operations.mode:
        raise ValueError("Snapshot has the wrong data mode")
    identity = required_text(raw.get("publication_id"), "publication_id")
    if identity.startswith("sim-") and operations.mode == "real":
        raise ValueError("Cannot import a simulated publication into real history")
    if identity not in by_id or by_id[identity][0]["status"] != "published":
        raise ValueError(f"Snapshot requires a confirmed publication: {identity}")
    post = by_id[identity][0]
    time = timestamp(raw.get("collected_at"), "collected_at")
    if time < timestamp(post["published_at"], "published_at") or operations.mode == "real" and time > now:
        raise ValueError("Snapshot must be collected between actual publication and now")
    values = {name: number(raw.get(name), name, integer=name in COUNTS,
                           money=name in MONEY, rate=name == "completion_rate") for name in METRICS}
    if all(value is None for value in values.values()):
        raise ValueError("Supply at least one known metric")
    view_metric = optional_text(raw.get("view_metric"), "view_metric")
    if values["view_count"] is not None and view_metric is None:
        raise ValueError("Enter the platform's metric label as view_metric alongside view_count")
    if post["format"] != "video" and any(values[name] is not None for name in ("completion_rate", "avg_watch_seconds")):
        raise ValueError("Completion rate and average watch time are video metrics")
    return {
        "schema_version": 1, "data_mode": operations.mode,
        "publication_id": identity, "collected_at": time.isoformat(),
        "view_metric": view_metric,
        "metric_scope": optional_text(raw.get("metric_scope"), "metric_scope") or "作品累计",
        **values,
    }


def import_metrics(args, operations):
    _, _, by_id = operations.publications()
    rows = operations.load(operations.snapshots)
    indexed = {(row["publication_id"], row["collected_at"]): row for row in rows}
    added = 0
    unchanged = 0
    now = datetime.now(timezone.utc)
    for line, raw in input_rows(args.file):
        try:
            row = snapshot(raw, operations, by_id, now)
            key = row["publication_id"], row["collected_at"]
            if key in indexed:
                if indexed[key] != row:
                    raise ValueError(f"Conflicting snapshot for {key[0]} at {key[1]}")
                unchanged += 1
                continue
            indexed[key] = row
            added += 1
        except (ValueError, TypeError) as error:
            raise ValueError(f"{args.file.name}, line {line}: {error}") from error
    if added:
        operations.directory.mkdir(parents=True, exist_ok=True)
        write_jsonl(operations.snapshots, indexed.values())
    return {"data_mode": operations.mode, "imported": added, "unchanged": unchanged}

WINDOWS = {24: 6, 72: 12, 168: 24}
RATES = {"like_rate": "likes", "save_rate": "saves", "share_rate": "shares"}
COMPARABLE = {
    "view_count": "播放／浏览", "new_followers_per_10k": "万次新增粉",
    "like_rate": "点赞率", "save_rate": "收藏率", "share_rate": "分享率",
    "completion_rate": "完播率", "avg_watch_seconds": "平均观看秒",
    "manual_minutes": "人工分钟", "production_cost_cny": "制作成本",
    "qualified_inquiries": "有效咨询", "revenue_cny": "收入", "total_cost_cny": "总成本", "net_cny": "净收益",
}


def publication_hours(post, snapshot):
    collected = timestamp(snapshot["collected_at"], "collected_at")
    published = timestamp(post["published_at"], "published_at")
    return (collected - published).total_seconds() / 3600


def calculated(post, snapshot):
    def ratio(numerator, factor=1):
        views = snapshot["view_count"]
        value = snapshot[numerator]
        return value * factor / views if views and value is not None else None

    production = post["production_cost_cny"]
    additional = snapshot["additional_cost_cny"]
    revenue = snapshot["revenue_cny"]
    total = Decimal(str(production)) + Decimal(str(additional)) if production is not None and additional is not None else None
    net = Decimal(str(revenue)) - total if total is not None and revenue is not None else None
    return {
        "publication_id": post["publication_id"], "platform": post["platform"], "format": post["format"],
        "lane": post["lane"], "category": post["category"], "title": post["title"],
        "duration_seconds": post["duration_seconds"],
        "collected_at": timestamp(snapshot["collected_at"], "collected_at").astimezone(SHANGHAI).isoformat(),
        "hours_after_publish": round(publication_hours(post, snapshot), 6),
        "view_metric": snapshot["view_metric"], "metric_scope": snapshot["metric_scope"],
        **{key: snapshot[key] for key in METRICS},
        "manual_minutes": post["manual_minutes"], "production_cost_cny": production,
        "new_followers_per_10k": ratio("new_followers", 10000),
        **{key: ratio(field) for key, field in RATES.items()},
        "total_cost_cny": float(total) if total is not None else None,
        "net_cny": float(net) if net is not None else None,
    }


def medians(rows):
    result = {}
    for key in COMPARABLE:
        values = [row[key] for row in rows if row[key] is not None]
        result[key] = {"n": len(values), "value": median(values) if values else None}
    return result


def comparisons(posts, snapshots):
    grouped = defaultdict(list)
    for post in posts:
        observations = [(publication_hours(post, row), row) for row in snapshots[post["publication_id"]]]
        for window, tolerance in WINDOWS.items():
            options = [(hours, row) for hours, row in observations if abs(hours - window) <= tolerance]
            if not options:
                continue
            _, chosen = min(options, key=lambda item: (abs(item[0] - window), item[0]))
            key = post["platform"], post["format"], window, chosen["view_metric"] or "", chosen["metric_scope"]
            grouped[key].append(calculated(post, chosen))
    result = []
    for (platform, form, window, label, scope), rows in sorted(grouped.items()):
        by_category = defaultdict(list)
        for row in rows:
            by_category[row["category"]].append(row)
        hours = [row["hours_after_publish"] for row in rows]
        result.append({
            "platform": platform, "format": form, "window_hours": window, "tolerance_hours": WINDOWS[window],
            "view_metric": label or None, "metric_scope": scope, "actual_hours_min": min(hours),
            "actual_hours_max": max(hours), "sample_works": len(rows), "medians": medians(rows),
            "categories": [{"category": category, "sample_works": len(works), "medians": medians(works)}
                           for category, works in sorted(by_category.items())],
            "works": rows,
        })
    return result


def review_period(args):
    now = datetime.now(SHANGHAI)
    as_of = timestamp(args.as_of, "as_of").astimezone(SHANGHAI) if args.as_of else now.replace(microsecond=0)
    if (args.start is None) != (args.through is None):
        raise ValueError("Provide both --start and --through, or use this week")
    if args.start is not None:
        start, through = date.fromisoformat(args.start), date.fromisoformat(args.through)
    else:
        start = as_of.date() - timedelta(days=as_of.weekday())
        through = start + timedelta(days=6)
    if start > through:
        raise ValueError("Start date must not be after through date")
    return start, through, as_of


def build_report(operations, start, through, as_of):
    _, _, all_records = operations.publications()
    records = [record for record, _ in all_records.values()]
    posts = []
    for row in records:
        if row["status"] != "published":
            continue
        published = timestamp(row["published_at"], "published_at").astimezone(SHANGHAI)
        if start <= published.date() <= through and published <= as_of:
            posts.append(row)
    posts.sort(key=lambda row: (row["platform"], row["format"], row["published_at"], row["publication_id"]))
    by_post = {row["publication_id"]: [] for row in posts}
    for row in operations.load(operations.snapshots):
        identity = row["publication_id"]
        if identity in by_post and timestamp(row["collected_at"], "collected_at") <= as_of:
            by_post[identity].append(row)
    latest = [calculated(post, max(by_post[post["publication_id"]],
                                   key=lambda row: timestamp(row["collected_at"], "collected_at")))
              for post in posts if by_post[post["publication_id"]]]
    facts = {
        "pending_records": sum(row["status"] == "pending" for row in records),
        "published_in_period": len(posts), "sample_works": len(latest),
        "eligible_snapshots": sum(map(len, by_post.values())),
    }
    data = {
        "schema_version": 1, "data_mode": operations.mode,
        "period": {"start": start.isoformat(), "through": through.isoformat(), "timezone": "Asia/Shanghai"},
        "facts": facts, "publications": posts,
        "observations": latest, "comparisons": comparisons(posts, by_post),
    }
    fingerprint = hashlib.sha256(json.dumps(data, ensure_ascii=False, sort_keys=True).encode("utf-8")).hexdigest()
    data["period"]["as_of"] = as_of.isoformat()
    data["basis_sha256"] = fingerprint
    return data


def editor_notes(path, data):
    if path is None:
        return []
    notes = read_json(path)
    if notes["data_mode"] != data["data_mode"] or notes["basis_sha256"] != data["basis_sha256"]:
        raise ValueError("Editorial notes do not match this report's data mode and basis_sha256")
    eligible = {row["publication_id"] for row in data["observations"]}
    result = []
    for item in notes["notes"]:
        refs = item["publication_ids"]
        if not isinstance(refs, list) or not refs or not set(refs) <= eligible:
            raise ValueError("Editorial notes must reference publications with eligible observations")
        result.append({"section": required_text(item["section"], "section"),
                       "text": required_text(item["text"], "text"), "publication_ids": refs})
    return result


def display(value, key=""):
    if value is None:
        return "缺失"
    if key in (*RATES, "completion_rate"):
        return f"{value:.2%}"
    if key.endswith("_cny"):
        return f"{value:,.2f}"
    if isinstance(value, float):
        return f"{value:,.2f}".rstrip("0").rstrip(".")
    if isinstance(value, int):
        return f"{value:,}"
    return str(value).replace("|", "｜").replace("\n", " ").replace("\r", " ")

    return "\n".join(lines) + "\n"


def report(args, operations):
    start, through, as_of = review_period(args)
    name = args.name or f"review-{start}-to-{through}"
    if not re.fullmatch(r"[a-z0-9][a-z0-9-]*", name):
        raise ValueError("Report name must use lowercase ASCII letters, digits and hyphens")
    data = build_report(operations, start, through, as_of)
    data["editorial_notes"] = editor_notes(args.editorial, data)
    dest = operations.directory / "reports" / name
    dest.mkdir(parents=True, exist_ok=True)
    atomic_write(dest / "review.json", json_text(data))
    atomic_write(dest / "review.md", markdown(data))
    return dest

### 发布记录

- `register` 读取 P7 清单，按其中已有的平台和形式分别创建 `pending` 记录。可用 `--platform douyin|xiaohongshu` 和 `--format video|note` 筛选。记录含 A/B 线、人工指定的内容分类、选题或片段 ID、清单路径和哈希、对应成品及视频时长。相同清单、平台和形式重试会返回同一记录。
- `confirm` 接受实际标题、带偏移的发布时间及 `--work-id` 或 `--url`。例如时间格式 `2026-11-26T15:30:00+08:00`。用 `--manual-minutes` 和 `--production-cost-cny` 添加工时和制作成本，只填写分配给该条发布记录的部分。还可用原确认参数重试以补齐工时、成本或作品链接。
- 已有 P5 选题的发布明细保存在该选题的 `publications` 数组中，首次实际发布确认与 `produced → published` 一起写入原选题历史。各平台和形式可处于不同状态。无 P5 选题的 A 线按片段关联，存入 `workspace/p8-operations/publications.jsonl`。旧的 P5 `published` 状态本身没有平台明细，需登记并确认实际作品才可导入运营快照。

### 数据录入

每行对应一条已确认作品在一次采集时的后台累计值。使用 UTF-8 CSV 或同名字段的 JSONL。空白或 JSON 的 `null` 表示缺失，后台确认的零填写 0。每行需要发布记录 ID、采集时间和至少一个已知指标。

| 字段 | 填法 |
| --- | --- |
| `publication_id`、`collected_at` | 本地发布记录 ID 和带时区偏移的 ISO 8601 采集时间 |
| `view_metric`、`view_count` | 照录后台名，如「播放量」或「浏览量」。填写数量时必填指标名称 |
| `likes`、`comments`、`saves`、`shares`、`new_followers` | 累计点赞、评论、收藏、分享、该作品新增粉丝 |
| `completion_rate`、`avg_watch_seconds` | 视频适用。完播率可填 `0.42` 或 `42%`，平均观看时长填秒数 |
| `impressions`、`qualified_inquiries` | 可选的曝光量及能归因到该作品的有效商业咨询数 |
| `revenue_cny`、`additional_cost_cny` | 累计实收和制作成本之外的可归因费用，单位人民币元，最多两位小数 |
| `metric_scope` | 可选，默认「作品累计」。仅填作品级累计指标。有不同流量范围或统计口径时，用精确、一致的标签区分 |

同一发布 ID 及采集时刻的相同数据可以重复导入。冲突数据会报出行号并拒绝整批写入。累计快照存入 `workspace/p8-operations/snapshots.jsonl`。

### 周报口径

- 默认选北京时间本周发布的作品。可用 `--start YYYY-MM-DD --through YYYY-MM-DD` 指定含首尾的发布日期范围，用 `--as-of` 指定快照截止时间。期间发布且截至时刻有有效快照的作品计入有效样本。
- 单条近况取截止时刻最新快照。分组比较按平台、形式、指标口径和发布后目标窗口进行，取各作品最接近目标的快照，同距取较早一笔。24、72、168 小时窗口分别允许偏差 6、12、24 小时，并显示实际发布后小时。累计快照逐笔保留，计算时每条作品每个窗口只用一笔。
- 每万次播放／浏览新增粉丝、点赞率、收藏率及分享率仅在对应分子已知且分母为正时计算。同一分组按内容分类计算中位数，并给出各指标有效样本数。已知制作成本及额外成本时计算总成本，再结合已知收入计算净收益。
- 报告写入 `workspace/p8-operations/reports/`。可用 `--name` 指定报告目录名。机器可读的 `review.json` 保留作品来源、最新快照、窗口明细和分组计算。Codex 按 Skill 用 `--editorial` 将注明依据的编辑判断写入 Markdown 与 JSON。运营建议保留给 GPT 或用户审核。
- 模拟入口为 `src/review.py --simulate <子命令>`，台账、模拟清单及报告均在 `workspace/p8-simulation/`，记录 ID 以 `sim-` 开头。P8 验收与示例见 [P8 审核文本](P8-REVIEW.txt)。

@contextmanager
def history_lock(history_path=HISTORY):
    history_path.parent.mkdir(parents=True, exist_ok=True)
    with history_path.with_suffix(".lock").open("a+b") as handle:
        if os.name == "nt":
            import msvcrt
            handle.seek(0)
            msvcrt.locking(handle.fileno(), msvcrt.LK_NBLCK, 1)
        else:
            import fcntl
            fcntl.flock(handle, fcntl.LOCK_EX | fcntl.LOCK_NB)
        # Closing the handle releases the lock, including after an interrupted process.
        yield

def read_history(history_path=HISTORY):
    if not history_path.exists():
        return []
    with history_path.open(encoding="utf-8") as source:
        return [json.loads(line) for line in source if line.strip()]

def write_history(history, history_path=HISTORY):
    atomic_write(history_path, "".join(json.dumps(row, ensure_ascii=False) + "\n" for row in history))

def history_status(args):
    if not args.topic_id:
        for row in read_history():
            print(json.dumps(row, ensure_ascii=False))
        return
    if not args.status or not args.note or not args.note.strip():
        raise ValueError("Updating history requires a status and a note recording the actual progress")
    with history_lock():
        history = read_history()
        row = next((row for row in history if row["topic_id"] == args.topic_id), None)
        if row is None:
            raise ValueError(f"Unknown topic: {args.topic_id}")
        if not (task_path(row["task"]) / "topics.json").is_file():
            raise ValueError("Finalize the research task before updating a topic status")
        if args.status not in TOPIC_STATES[row["status"]]:
            raise ValueError(f"Invalid status transition: {row['status']} to {args.status}")
        if args.status == "published" and row.get("publications"):
            raise ValueError("Confirm the actual platform publication with src/review.py confirm")
        row.update(status=args.status, status_note=args.note.strip(),
                   status_at=datetime.now(SHANGHAI).isoformat(timespec="seconds"))
        write_history(history)
    print(f"{args.topic_id}: {args.status}")
